import {createCharacter,getScene,presentChoices,commitChoice,sceneText}from'/workspace/saint-shard-fix/lib/engine';
import{visibleCampaignDialogue}from'/workspace/saint-shard-fix/lib/campaign-dialogue';
import{sceneIllustration,portraitArt}from'/workspace/saint-shard-fix/lib/art';
import{readFileSync,writeFileSync,readdirSync}from'node:fs';
import{createHash}from'node:crypto';
import{execFileSync}from'node:child_process';
const files=(d:string):string[]=>readdirSync(d,{withFileTypes:true}).flatMap(r=>r.isDirectory()?files(`${d}/${r.name}`):[`${d}/${r.name}`]);
const sha=(b:string|Buffer)=>createHash('sha256').update(b).digest('hex');
const rows=[];const routes=files('qa/routes').filter(f=>f.endsWith('.json')).sort();
for(const file of routes){const spec=JSON.parse(readFileSync(file,'utf8'));let state=createCharacter(spec.character);for(let step=0;step<=spec.steps.length;step++){
 const scene=getScene(state.sceneId),text=sceneText(scene,state),choices=presentChoices(state,scene);
 rows.push({file,step,state,text,choices,voice:visibleCampaignDialogue(scene.id,scene.speaker,text),actor:scene.speaker,art:sceneIllustration(scene.id,state.flags),portrait:portraitArt(scene.speaker,scene.id,state.flags)});
 if(step===spec.steps.length)break;const action=spec.steps[step],choice=choices.find(c=>c.id===action.choiceId&&c.enabled)!;state=commitChoice(state,choice,choice.check?{roll:action.roll}:undefined).state;
}}
const media=[...files('public/art'),...files('public/audio')].sort().map(path=>({path,sha256:sha(readFileSync(path))}));
const baseline={version:1,phase:21,revision:execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim(),routes:routes.length,routeFiles:routes,visits:rows.length,campaignSha256:sha(JSON.stringify(rows)),media,manifests:files('qa/media').filter(f=>f.endsWith('.json')).sort().map(path=>({path,sha256:sha(readFileSync(path))}))};
writeFileSync('/tmp/saint-phase21/baseline.json',JSON.stringify(baseline,null,2)+'\n');console.log(JSON.stringify({routes:routes.length,visits:rows.length,hash:baseline.campaignSha256,media:media.length}));
