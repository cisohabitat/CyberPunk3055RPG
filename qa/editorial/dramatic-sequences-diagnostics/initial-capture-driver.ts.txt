import{chromium}from'/workspace/saint-shard-fix/node_modules/playwright';import{mkdirSync,writeFileSync,readFileSync}from'node:fs';import{createHash}from'node:crypto';import{execFileSync}from'node:child_process';import{SEQUENCE_CHECKPOINTS}from'/workspace/saint-shard-fix/qa/production/dramatic-sequences-checkpoints';
async function main(){
const out='/workspace/saint-shard-fix/qa/production/reports/dramatic-sequences-captures';mkdirSync(out);const browser=await chromium.launch({executablePath:'/usr/bin/chromium'});const rows=[];
try{for(const{rule,state}of SEQUENCE_CHECKPOINTS)for(const mode of ['before','current']){const s=structuredClone(state);if(mode==='before')delete s.flags.dramatic_sequence_art_enabled;const page=await browser.newPage({viewport:{width:390,height:900},reducedMotion:'reduce'});await page.addInitScript(s=>{localStorage.setItem('saint-shard-3055-v1',JSON.stringify(s));localStorage.setItem('saint-shard-preferences',JSON.stringify({sound:false,artwork:'full',reading:'all',textStep:2,motion:'reduce'}));},s);await page.goto('http://127.0.0.1:3128');await page.getByTestId('continue-run').click();const im=page.getByTestId('scene-illustration').locator('img');await im.scrollIntoViewIfNeeded();await im.evaluate((i:HTMLImageElement)=>i.decode());await page.evaluate(()=>scrollTo(0,0));const file=out+'/'+rule.id+'-'+mode+'.png';await page.screenshot({path:file,fullPage:true});rows.push({id:rule.id,mode,sceneId:s.sceneId,source:await im.getAttribute('src'),file,sha256:createHash('sha256').update(readFileSync(file)).digest('hex')});await page.close();}}finally{await browser.close();}
writeFileSync('qa/editorial/dramatic-sequences-captures.json',JSON.stringify({revision:execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim(),dirty:true,method:'Same current Chromium source, legal route checkpoints; preceding supported visual flag absent for before, present for current. Enlarged390px, full-scene reading, actual image decode. This does not relabel source as the preceding deployed renderer. All preceding source selections independently preserved by baseline.',rows},null,2)+'\n');console.log(rows.length+' paired captures');

}
main().catch(e=>{console.error(e);process.exitCode=1});
